## responses 端点格式（探针已验证，照此实现）
- 请求：`POST {BASE}/v1/responses`，body `{model, input:[{role,content}], tools:[...], max_output_tokens}`
- tools 格式（**扁平**，不是 chat 的嵌套）：function → `{type:'function', name, description, parameters}`；联网 → `{type:'web_search'}`
- 响应：`output[]` 数组，项 `type` 为 `message`（content `[{type:'output_text',text}]`）/ `function_call`（`{name, arguments, call_id}`）/ `web_search_call`（号池已执行，结果在 message）
- 多轮工具回传：function 结果作为 input 续传项 `{type:'function_call_output', call_id, output}`
- usage：`input_tokens` / `output_tokens`

`server/openaiResponsesProvider.mjs`：

```js
const DEFAULT_TIMEOUT_MS = Number(process.env.OPENAI_COMPATIBLE_TIMEOUT_MS || 240000);

const getConfig = (env = {}) => ({
  apiKey: String(env.OPENAI_COMPATIBLE_API_KEY || '').trim(),
  baseUrl: String(env.OPENAI_COMPATIBLE_BASE_URL || 'https://maxforai.top').trim().replace(/\/$/, ''),
  allowedModels: String(env.OPENAI_COMPATIBLE_MODELS || '').split(',').map((s) => s.trim()).filter(Boolean),
});

// chat 工具(嵌套 function / web_search) → responses 扁平格式
export const toResponsesTool = (tool) => {
  if (tool?.type === 'function' && tool.function) {
    return { type: 'function', name: tool.function.name, description: tool.function.description, parameters: tool.function.parameters };
  }
  return tool; // web_search / 已是扁平 function 等原样
};

export const parseResponsesOutput = (data = {}) => {
  const items = Array.isArray(data?.output) ? data.output : [];
  let content = '';
  const toolCalls = [];
  for (const item of items) {
    if (item.type === 'message') {
      content += (item.content || []).map((c) => c.text || '').join('');
    } else if (item.type === 'function_call') {
      let args;
      try { args = JSON.parse(String(item.arguments || '{}')); } catch { continue; }
      toolCalls.push({ id: String(item.call_id || item.id || ''), name: String(item.name || ''), args });
    }
    // web_search_call 等内置工具:号池已执行,结果并入 message,无需处理
  }
  return {
    content: content.trim(),
    toolCalls,
    finishReason: toolCalls.length > 0 ? 'tool_calls' : 'stop',
    usage: data?.usage || {},
  };
};

// messages([{role,content}]) → responses input;附带工具结果项
export const runResponsesJob = async ({ payload = {}, env = {}, signal = null } = {}) => {
  const { apiKey, baseUrl, allowedModels } = getConfig(env);
  if (!apiKey) { const e = new Error('OPENAI_COMPATIBLE_API_KEY 未配置'); e.code = 'provider_config_missing'; throw e; }
  const model = String(payload?.model || '').trim();
  if (allowedModels.length > 0 && !allowedModels.includes(model)) { const e = new Error(`模型 ${model} 不在白名单`); e.code = 'provider_bad_request'; throw e; }

  const input = Array.isArray(payload?.input) ? payload.input : (Array.isArray(payload?.messages) ? payload.messages : []);
  const tools = (Array.isArray(payload?.tools) ? payload.tools : []).map(toResponsesTool);
  const body = {
    model,
    input,
    ...(tools.length > 0 ? { tools } : {}),
    ...(payload?.maxTokens ? { max_output_tokens: Number(payload.maxTokens) } : {}),
  };

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), DEFAULT_TIMEOUT_MS);
  if (signal) signal.addEventListener('abort', () => controller.abort(), { once: true });
  try {
    const res = await fetch(`${baseUrl}/v1/responses`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${apiKey}` },
      body: JSON.stringify(body),
      signal: controller.signal,
    });
    if (!res.ok) {
      const t = await res.text().catch(() => '');
      const e = new Error(`responses 请求失败 (${res.status}): ${t.slice(0, 200)}`);
      e.code = res.status === 401 || res.status === 403 ? 'provider_auth_invalid' : 'provider_bad_response';
      throw e;
    }
    const data = await res.json().catch(() => ({}));
    return { ...parseResponsesOutput(data), modelUsed: model, raw: data };
  } finally {
    clearTimeout(timer);
  }
};
```

```js
const MAX_TOOL_ROUNDS = Number(process.env.AGENT_TOOL_MAX_ROUNDS || 5);
let imageGenerated = false; // 生图限一次
let rounds = 0;
let response = first;
const convo = [...messages];

while (response.finishReason === 'tool_calls' && response.toolCalls?.length && rounds < MAX_TOOL_ROUNDS) {
  rounds += 1;
  const call = response.toolCalls.find((c) => ['generate_image', 'search_knowledge'].includes(c.name)) || response.toolCalls[0];
  // 把 assistant 的工具调用加进对话
  convo.push({ role: 'assistant', content: response.content || null, tool_calls: [{ id: call.id || `call_${rounds}`, type: 'function', function: { name: call.name, arguments: JSON.stringify(call.args) } }] });

  let toolOutput;
  if (call.name === 'search_knowledge') {
    emit('searching_knowledge', { query: call.args?.query });
    try {
      const { query } = normalizeSearchKnowledgeArgs(call.args);
      const chunks = typeof searchKnowledge === 'function' ? await searchKnowledge(query) : [];
      toolOutput = chunks.length
        ? `检索结果:\n${chunks.map((c, i) => `资料${i + 1}(${c.documentTitle || c.sourceType || '知识'}): ${c.content}`).join('\n\n')}`
        : '未在知识库中找到相关内容。';
    } catch (error) {
      toolOutput = `知识库检索失败: ${error?.message || '未知错误'}。请据此向用户说明,不要编造。`;
    }
  } else if (call.name === 'generate_image') {
    if (imageGenerated) { toolOutput = '已生成过图片,不再重复生成。'; }
    else {
      // ... 复用现有 generate_image 执行逻辑(emit image_generating, 调 generateImage, validInputUrls 校验, imagePlan 记录) ...
      imageGenerated = true;
    }
  } else {
    toolOutput = '不支持的工具。';
  }
  convo.push({ role: 'tool', tool_call_id: call.id || `call_${rounds}`, content: toolOutput });

  response = await callModel({ messages: convo, tools, toolChoice: 'auto', maxTokens: contextLimits.maxOutputTokens, onDelta: (d) => emit('streaming', { delta: d }) });
}
emit('done', {});
// 返回 response.content + 累积的 imagePlan/imageResultUrls
```

注意：保留第2期已有的 generate_image 细节（buildUserMessageContent 上传图、validInputUrls 校验、imagePlan 构造、freshUploadGuidance）。本 Task 是把"单次 generate_image"扩成"工具循环"，generate_image 分支内部逻辑不变，只是搬进循环并加 `imageGenerated` 守卫。
