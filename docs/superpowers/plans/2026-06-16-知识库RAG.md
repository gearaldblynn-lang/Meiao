const DEFAULT_BASE_URL = 'https://ark.cn-beijing.volces.com/api/v3';
const DEFAULT_MODEL = 'doubao-embedding-vision-251215';
const DEFAULT_TIMEOUT_MS = Number(process.env.EMBEDDING_TIMEOUT_MS || 30000);

const getDoubaoConfig = (env = {}) => ({
  apiKey: String(env.DOUBAO_EMBEDDING_API_KEY || '').trim(),
  baseUrl: String(env.DOUBAO_EMBEDDING_BASE_URL || DEFAULT_BASE_URL).trim().replace(/\/$/, ''),
  model: String(env.DOUBAO_EMBEDDING_MODEL || DEFAULT_MODEL).trim(),
});

// 单条文本 → 豆包 multimodal embedding。豆包 multimodal 端点一次算一条 input。
const embedOneDoubao = async (text, { apiKey, baseUrl, model }, signal) => {
  const res = await fetch(`${baseUrl}/embeddings/multimodal`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${apiKey}` },
    body: JSON.stringify({ model, input: [{ type: 'text', text: String(text || '') }] }),
    signal,
  });
  if (!res.ok) {
    const t = await res.text().catch(() => '');
    const e = new Error(`embedding 请求失败 (${res.status}): ${t.slice(0, 200)}`);
    e.code = 'embedding_provider_error';
    throw e;
  }
  const data = await res.json().catch(() => ({}));
  // multimodal 返回 data.embedding（单条）或 data[0].embedding
  const vec = data?.data?.embedding || data?.data?.[0]?.embedding;
  if (!Array.isArray(vec)) {
    const e = new Error('embedding 响应缺少向量');
    e.code = 'embedding_provider_error';
    throw e;
  }
  return vec;
};

// 对外统一接口：一批文本 → 一批向量（顺序对齐）。可插拔：按 EMBEDDING_PROVIDER 分发。
export const embedTexts = async (texts, env = {}) => {
  const list = Array.isArray(texts) ? texts : [texts];
  const provider = String(env.EMBEDDING_PROVIDER || 'doubao').trim();
  if (provider !== 'doubao') {
    const e = new Error(`不支持的 embedding provider: ${provider}`);
    e.code = 'embedding_provider_error';
    throw e;
  }
  const cfg = getDoubaoConfig(env);
  if (!cfg.apiKey) {
    const e = new Error('DOUBAO_EMBEDDING_API_KEY 未配置');
    e.code = 'embedding_config_missing';
    throw e;
  }
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), DEFAULT_TIMEOUT_MS);
  try {
    const out = [];
    for (const text of list) {
      out.push(await embedOneDoubao(text, cfg, controller.signal));
    }
    return out;
  } finally {
    clearTimeout(timer);
  }
};
```

export const cosineSimilarity = (a, b) => {
  if (!Array.isArray(a) || !Array.isArray(b) || a.length !== b.length || a.length === 0) return 0;
  let dot = 0, na = 0, nb = 0;
  for (let i = 0; i < a.length; i += 1) {
    const x = Number(a[i]) || 0;
    const y = Number(b[i]) || 0;
    dot += x * y; na += x * x; nb += y * y;
  }
  if (na === 0 || nb === 0) return 0;
  return dot / (Math.sqrt(na) * Math.sqrt(nb));
};

// queryVec: 数组；chunks: [{id, content, embedding, ...}]；policy: {topK, maxChunks, maxContextChars, minSimilarity}
export const rankChunksByVector = (queryVec, chunks, policy = {}) => {
  const topK = Number(policy.topK || 3);
  const maxChunks = Number(policy.maxChunks || 5);
  const maxContextChars = Number(policy.maxContextChars || 2400);
  const minSimilarity = Number(policy.minSimilarity || 0);

  const scored = (Array.isArray(chunks) ? chunks : [])
    .filter((c) => Array.isArray(c?.embedding) && c.embedding.length > 0)
    .map((c) => ({ ...c, score: cosineSimilarity(queryVec, c.embedding) }))
    .filter((c) => c.score >= minSimilarity)
    .sort((a, b) => b.score - a.score || Number(a.chunkIndex || 0) - Number(b.chunkIndex || 0));

  const out = [];
  let totalChars = 0;
  for (const c of scored) {
    if (out.length >= maxChunks || out.length >= topK) break;
    const content = String(c.content || '');
    if (!content) continue;
    if (totalChars + content.length > maxContextChars) break;
    out.push(c);
    totalChars += content.length;
  }
  return out;
};
```

```js
const CHUNK_OVERLAP_RATIO = 0.15;

const splitChunkByLength = (value, maxChunkChars) => {
  const source = String(value || '').trim();
  if (!source) return [];
  if (source.length <= maxChunkChars) return [source];

  const overlap = Math.min(
    Math.floor(maxChunkChars * CHUNK_OVERLAP_RATIO),
    Math.floor(maxChunkChars / 3)
  );
  const slices = [];
  let start = 0;
  while (start < source.length) {
    let end = Math.min(start + maxChunkChars, source.length);
    // 若未到结尾,尽量在句子边界断（向前找最近的句末标点）
    if (end < source.length) {
      const window = source.slice(start, end);
      const lastBoundary = Math.max(
        window.lastIndexOf('。'), window.lastIndexOf('！'), window.lastIndexOf('？'),
        window.lastIndexOf('!'), window.lastIndexOf('?'), window.lastIndexOf('\n')
      );
      // 边界要足够靠后（过半）才采用,否则块太短,回退硬切
      if (lastBoundary >= maxChunkChars * 0.5) {
        end = start + lastBoundary + 1;
      }
    }
    const slice = source.slice(start, end).trim();
    if (slice) slices.push(slice);
    if (end >= source.length) break;
    // 下一块起点回退 overlap,制造重叠
    start = Math.max(end - overlap, start + 1);
  }
  return slices;
};
```

import { embedTexts } from './embeddingProvider.mjs';
import { rankChunksByVector } from './vectorSearch.mjs';

// 纯向量检索；embedding 不可用/无配置/失败 → 降级回关键词 fallbackSearch。
export const searchKnowledgeChunksByVector = async (query, chunks, policy = {}, env = {}, fallbackSearch = null) => {
  const hasKey = String(env.DOUBAO_EMBEDDING_API_KEY || '').trim();
  if (hasKey) {
    try {
      const [queryVec] = await embedTexts([query], env);
      const ranked = rankChunksByVector(queryVec, chunks, {
        topK: Number(policy.topK || 3),
        maxChunks: Number(policy.maxChunks || 5),
        maxContextChars: Number(policy.maxContextChars || 2400),
        minSimilarity: Number(env.EMBEDDING_MIN_SIMILARITY || policy.similarityThreshold || 0.3),
      });
      if (ranked.length > 0) return ranked;
      // 向量召回为空(可能全是旧块无向量)→ 降级关键词
    } catch (error) {
      console.warn('[rag] 向量检索失败,降级关键词', { message: error?.message || String(error) });
    }
  }
  if (typeof fallbackSearch === 'function') return fallbackSearch(chunks, query, policy);
  return [];
};
